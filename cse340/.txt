const getStudentCourses = async (studentId) => {
    const query = `
        SELECT c.title, e.grade
        FROM Enrollments e
        JOIN Courses c ON e.course_id = c.course_id
        WHERE e.student_id = $1

    const result = await db.query(query, [studentId]);
